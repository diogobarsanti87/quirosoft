import { useState, useEffect } from 'react';
import axios from 'axios';
import { Calendar as BigCalendar, dateFnsLocalizer } from 'react-big-calendar';
import { format, parse, startOfWeek, getDay } from 'date-fns';
import ptBR from 'date-fns/locale/pt-BR';
import 'react-big-calendar/lib/css/react-big-calendar.css';
import { UserPlus, Users, Calendar as CalendarIcon, Phone, Clock, CheckCircle, XCircle } from 'lucide-react';
import logoImg from './assets/logo.png';

const locales = {
  'pt-BR': ptBR,
};

const localizer = dateFnsLocalizer({
  format,
  parse,
  startOfWeek: () => startOfWeek(new Date(), { weekStartsOn: 1 }),
  getDay,
  locales,
});

function App() {
  const [abaAtiva, setAbaAtiva] = useState('agendamentos');
  const [pacientes, setPacientes] = useState([]);
  const [agendamentos, setAgendamentos] = useState([]);

  // Form de Paciente
  const [formPaciente, setFormPaciente] = useState({
    nome: '', cpf: '', telefone: '', data_nascimento: ''
  });

  // Form de Agendamento
  const [formAgendamento, setFormAgendamento] = useState({
    paciente_id: '', data_hora: '', observacoes: ''
  });

  const API_PACIENTES = 'http://localhost:3001/api/pacientes';
  const API_AGENDAMENTOS = 'http://localhost:3001/api/agendamentos';

  const fetchPacientes = async () => {
    try {
      const res = await axios.get(API_PACIENTES);
      setPacientes(res.data);
    } catch (err) {
      console.error('Erro ao buscar pacientes:', err);
    }
  };

  const fetchAgendamentos = async () => {
    try {
      const res = await axios.get(API_AGENDAMENTOS);
      setAgendamentos(res.data);
    } catch (err) {
      console.error('Erro ao buscar agendamentos:', err);
    }
  };

  useEffect(() => {
    fetchPacientes();
    fetchAgendamentos();
  }, []);

  const handleSalvarPaciente = async (e) => {
    e.preventDefault();
    try {
      await axios.post(API_PACIENTES, formPaciente);
      setFormPaciente({ nome: '', cpf: '', telefone: '', data_nascimento: '' });
      fetchPacientes();
    } catch (err) {
      console.error('Erro ao cadastrar paciente:', err);
    }
  };

  const handleSalvarAgendamento = async (e) => {
    e.preventDefault();
    try {
      await axios.post(API_AGENDAMENTOS, formAgendamento);
      setFormAgendamento({ paciente_id: '', data_hora: '', observacoes: '' });
      fetchAgendamentos();
    } catch (err) {
      console.error('Erro ao criar agendamento:', err);
    }
  };

  const atualizarStatus = async (id, status) => {
    try {
      await axios.patch(`${API_AGENDAMENTOS}/${id}/status`, { status });
      fetchAgendamentos();
    } catch (err) {
      console.error('Erro ao atualizar status:', err);
    }
  };

  // Mapeia agendamentos para o formato do BigCalendar
  const eventos = agendamentos.map((a) => {
    const inicio = new Date(a.data_hora);
    const fim = new Date(inicio.getTime() + 60 * 60 * 1000); // Duração de 1 hora
    return {
      id: a.id,
      title: `${a.paciente_nome} (${a.observacoes || 'Consulta'})`,
      start: inicio,
      end: fim,
      status: a.status
    };
  });

  const eventStyleGetter = (event) => {
    let backgroundColor = '#007bff';
    if (event.status === 'realizado') backgroundColor = '#28a745';
    if (event.status === 'cancelado') backgroundColor = '#dc3545';
    return {
      style: {
        backgroundColor,
        borderRadius: '5px',
        color: 'white',
        border: 'none',
        display: 'block',
        fontSize: '12px',
        padding: '2px 5px'
      }
    };
  };

  return (
    <div style={{ fontFamily: 'sans-serif', padding: '20px', maxWidth: '1200px', margin: '0 auto' }}>
      
      {/* LOGO EM FORMATO DE CÍRCULO */}
      <div style={{ display: 'flex', justifyContent: 'left', marginBottom: '10px' }}>
        <div
          style={{
            width: '115px',
            height: '90px',
            borderRadius: '50%',
            overflow: 'hidden',
            display: 'flex',
            alignItems: 'center',
            justifyContent: 'center',
            boxShadow: '0 8px 10px rgba(0,0,0,0.15)'
          }}
        >
          <img
            src={logoImg}
            alt="Logo Quirosoft"
            style={{
              width: '150%',
              height: '150%',
              objectFit: 'cover'
            }}
          />
        </div>
      </div>

      {/* Menu de Abas */}
      <div style={{ display: 'flex', gap: '10px', marginBottom: '20px' }}>
        <button
          onClick={() => setAbaAtiva('pacientes')}
          style={{
            padding: '10px 20px', cursor: 'pointer', borderRadius: '5px', border: 'none',
            backgroundColor: abaAtiva === 'pacientes' ? '#007bff' : '#e0e0e0',
            color: abaAtiva === 'pacientes' ? 'white' : 'black'
          }}
        >
          <Users size={16} /> Pacientes
        </button>
        <button
          onClick={() => setAbaAtiva('agendamentos')}
          style={{
            padding: '10px 20px', cursor: 'pointer', borderRadius: '5px', border: 'none',
            backgroundColor: abaAtiva === 'agendamentos' ? '#007bff' : '#e0e0e0',
            color: abaAtiva === 'agendamentos' ? 'white' : 'black'
          }}
        >
          <CalendarIcon size={16} /> Agendamentos
        </button>
      </div>

      {/* ABA PACIENTES */}
      {abaAtiva === 'pacientes' && (
        <div>
          <div style={{ border: '1px solid #ccc', padding: '20px', borderRadius: '8px', marginBottom: '30px' }}>
            <h2><UserPlus size={20} /> Cadastrar Paciente</h2>
            <form onSubmit={handleSalvarPaciente} style={{ display: 'grid', gap: '10px', gridTemplateColumns: '1fr 1fr' }}>
              <input
                type="text" placeholder="Nome Completo" value={formPaciente.nome}
                onChange={(e) => setFormPaciente({ ...formPaciente, nome: e.target.value })} required style={{ padding: '8px' }}
              />
              <input
                type="text" placeholder="CPF" value={formPaciente.cpf}
                onChange={(e) => setFormPaciente({ ...formPaciente, cpf: e.target.value })} style={{ padding: '8px' }}
              />
              <input
                type="text" placeholder="Telefone" value={formPaciente.telefone}
                onChange={(e) => setFormPaciente({ ...formPaciente, telefone: e.target.value })} style={{ padding: '8px' }}
              />
              <input
                type="date" value={formPaciente.data_nascimento}
                onChange={(e) => setFormPaciente({ ...formPaciente, data_nascimento: e.target.value })} style={{ padding: '8px' }}
              />
              <button type="submit" style={{ gridColumn: 'span 2', padding: '10px', backgroundColor: '#007bff', color: 'white', border: 'none', borderRadius: '4px', cursor: 'pointer' }}>
                Salvar Paciente
              </button>
            </form>
          </div>

          <h2><Users size={20} /> Lista de Pacientes</h2>
          <ul style={{ listStyle: 'none', padding: 0 }}>
            {pacientes.map((p) => (
              <li key={p.id} style={{ border: '1px solid #eee', padding: '15px', borderRadius: '6px', marginBottom: '10px', display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}>
                <div>
                  <strong>{p.nome}</strong> <br />
                  <small><Phone size={14} /> {p.telefone || 'Sem telefone'} | CPF: {p.cpf || 'N/A'}</small>
                </div>
              </li>
            ))}
          </ul>
        </div>
      )}

      {/* ABA AGENDAMENTOS */}
      {abaAtiva === 'agendamentos' && (
        <div>
          <div style={{ border: '1px solid #ccc', padding: '20px', borderRadius: '8px', marginBottom: '30px' }}>
            <h2><CalendarIcon size={20} /> Agendar Consulta</h2>
            <form onSubmit={handleSalvarAgendamento} style={{ display: 'grid', gap: '10px', gridTemplateColumns: '1fr 1fr' }}>
              <select
                value={formAgendamento.paciente_id}
                onChange={(e) => setFormAgendamento({ ...formAgendamento, paciente_id: e.target.value })}
                required style={{ padding: '8px' }}
              >
                <option value="">Selecione um Paciente...</option>
                {pacientes.map((p) => (
                  <option key={p.id} value={p.id}>{p.nome}</option>
                ))}
              </select>

              <input
                type="datetime-local"
                value={formAgendamento.data_hora}
                onChange={(e) => setFormAgendamento({ ...formAgendamento, data_hora: e.target.value })}
                required style={{ padding: '8px' }}
              />

              <input
                type="text"
                placeholder="Observações (ex: Primeira consulta, dor lombar)"
                value={formAgendamento.observacoes}
                onChange={(e) => setFormAgendamento({ ...formAgendamento, observacoes: e.target.value })}
                style={{ gridColumn: 'span 2', padding: '8px' }}
              />

              <button type="submit" style={{ gridColumn: 'span 2', padding: '10px', backgroundColor: '#28a745', color: 'white', border: 'none', borderRadius: '4px', cursor: 'pointer' }}>
                Confirmar Agendamento
              </button>
            </form>
          </div>

          {/* GRADE HORÁRIA VISUAL (08:00 ÀS 22:00) */}
          <div style={{ height: '650px', backgroundColor: '#fff', padding: '15px', borderRadius: '8px', border: '1px solid #ccc', marginBottom: '30px' }}>
            <BigCalendar
              localizer={localizer}
              events={eventos}
              defaultView="week"
              views={['month', 'week', 'day']}
              min={new Date(2026, 0, 1, 8, 0)}
              max={new Date(2026, 0, 1, 22, 0)}
              step={30}
              timeslots={2}
              eventPropGetter={eventStyleGetter}
              messages={{
                next: 'Próximo',
                previous: 'Anterior',
                today: 'Hoje',
                month: 'Mês',
                week: 'Semana',
                day: 'Dia'
              }}
              culture="pt-BR"
            />
          </div>

          {/* LISTA E GERENCIAMENTO DE STATUS */}
          <h2><Clock size={20} /> Gerenciar Status das Consultas</h2>
          <ul style={{ listStyle: 'none', padding: 0 }}>
            {agendamentos.map((a) => (
              <li key={a.id} style={{ border: '1px solid #eee', padding: '15px', borderRadius: '6px', marginBottom: '10px', display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}>
                <div>
                  <strong>{a.paciente_nome}</strong> - <small>{a.telefone}</small><br />
                  <span>
                    <Clock size={14} /> {
                      new Date(a.data_hora).toLocaleString('pt-BR', {
                        weekday: 'long',
                        day: '2-digit',
                        month: '2-digit',
                        year: 'numeric',
                        hour: '2-digit',
                        minute: '2-digit'
                      })
                    }
                  </span> <br />
                  <small style={{ color: '#666' }}>Obs: {a.observacoes || 'Nenhuma'}</small> <br />
                  <span style={{ fontWeight: 'bold', color: a.status === 'realizado' ? 'green' : a.status === 'cancelado' ? 'red' : 'orange' }}>
                    Status: {a.status}
                  </span>
                </div>
                <div style={{ display: 'flex', gap: '5px' }}>
                  <button onClick={() => atualizarStatus(a.id, 'realizado')} style={{ backgroundColor: '#28a745', color: 'white', border: 'none', padding: '6px 10px', borderRadius: '4px', cursor: 'pointer' }}>
                    <CheckCircle size={14} /> Realizada
                  </button>
                  <button onClick={() => atualizarStatus(a.id, 'cancelado')} style={{ backgroundColor: '#dc3545', color: 'white', border: 'none', padding: '6px 10px', borderRadius: '4px', cursor: 'pointer' }}>
                    <XCircle size={14} /> Cancelar
                  </button>
                </div>
              </li>
            ))}
          </ul>
        </div>
      )}
    </div>
  );
}

export default App;
